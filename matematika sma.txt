<h3>1. Limit Fungsi</h3>
<p>Limit adalah nilai yang didekati fungsi ketika x mendekati suatu nilai.</p>
<p>Notasi: lim(x→a) f(x) = L</p>
<p>Contoh:</p>
<ul>
<li>lim(x→a) k = k</li>
<li>lim(x→a) x = a</li>
<li>lim(x→∞) 1/x = 0</li>
</ul>

<h3>2. Turunan</h3>
<p>Turunan menunjukkan laju perubahan fungsi.</p>
<p>Rumus dasar:</p>
<ul>
<li>d/dx(xⁿ) = nxⁿ⁻¹</li>
<li>d/dx(sin x) = cos x</li>
<li>d/dx(cos x) = -sin x</li>
</ul>
<p>Contoh: f(x) = 3x² → f'(x) = 6x</p>

<h3>3. Aplikasi Turunan</h3>
<ul>
<li>Menentukan nilai maksimum dan minimum</li>
<li>Menentukan interval naik/turun</li>
<li>Kecepatan dan percepatan</li>
</ul>

<h3>4. Integral</h3>
<p>Integral adalah kebalikan dari turunan.</p>
<p>Rumus dasar:</p>
<ul>
<li>∫ xⁿ dx = xⁿ⁺¹/(n+1) + C</li>
<li>∫ sin x dx = -cos x + C</li>
<li>∫ cos x dx = sin x + C</li>
</ul>

<h3>5. Integral Tentu</h3>
<p>Menghitung luas daerah:</p>
<p>∫ₐᵇ f(x) dx = F(b) - F(a)</p>

ctx.strokeStyle = '#999';
ctx.lineWidth = 2;
let cx = w/2;
let cy = h/2;
ctx.beginPath();
ctx.moveTo(0, cy);
ctx.lineTo(w, cy);
ctx.stroke();
ctx.beginPath();
ctx.moveTo(cx, 0);
ctx.lineTo(cx, h);
ctx.stroke();

// Plot
try {
ctx.strokeStyle = '#667eea';
ctx.lineWidth = 3;
ctx.beginPath();
let points = [];
for(let px = 0; px < w; px++){
let x = xMin + (px/w)*(xMax-xMin);
let y = eval(func.replace(/x/g, '('+x+')'));
let py = cy - (y*20);
points.push({x:px, y:py});
}
ctx.moveTo(points[0].x, points[0].y);
for(let i = 1; i < points.length; i++){
ctx.lineTo(points[i].x, points[i].y);
}
ctx.stroke();
addActivity('Membuat grafik');
} catch(e) {
alert('Error dalam fungsi!');
}
}

// Quiz
function jawab(btn, benar, id){
stats.total++;
let hasil = document.getElementById(id);
if(benar){
hasil.innerHTML = '✅ Benar!';
hasil.style.color = '#10b981';
btn.classList.add('correct');
stats.benar++;
} else {
hasil.innerHTML = '❌ Salah';
hasil.style.color = '#ef4444';
btn.classList.add('wrong');
}
btn.parentElement.querySelectorAll('button').forEach(b => b.disabled = true);
updateStats();
addActivity('Menjawab soal ' + (benar ? 'benar' : 'salah'));
}

function cekJawaban(inputId, jawaban){
let input = document.getElementById(inputId);
let hasil = document.getElementById('h-'+inputId);
let nilai = parseFloat(input.value);
stats.total++;
if(nilai === jawaban){
hasil.innerHTML = '✅ Benar!';
hasil.style.color = '#10b981';
stats.benar++;
} else {
hasil.innerHTML = '❌ Salah. Jawaban: ' + jawaban;
hasil.style.color = '#ef4444';
}
updateStats();
addActivity('Latihan interaktif');
}

}

// Stats
function updateStats(){
document.getElementById('totalSoal').textContent = stats.total;
document.getElementById('soalBenar').textContent = stats.benar;
let acc = stats.total > 0 ? Math.round((stats.benar/stats.total)*100) : 0;
document.getElementById('akurasi').textContent = acc + '%';
localStorage.setItem('stats', JSON.stringify(stats));
}

function updateTaskCount(){
let count = 0;
for(let i = 1; i <= 3; i++){
if(localStorage.getItem('t10-'+i)) count++;
if(localStorage.getItem('t11-'+i)) count++;
if(localStorage.getItem('t12-'+i)) count++;
}
document.getElementById('tugasSelesai').textContent = count;
}

function addActivity(text){
let log = document.getElementById('activityLog');
let now = new Date();
let time = now.getHours() + ':' + String(now.getMinutes()).padStart(2,'0');
let activities = JSON.parse(localStorage.getItem('activities') || '[]');
activities.unshift({time:time, text:text});
activities = activities.slice(0,5);
localStorage.setItem('activities', JSON.stringify(activities));
if(activities.length > 0){
log.innerHTML = activities.map(a =>
'<div class="activity-item"><small>'+a.time+'</small> - '+a.text+'</div>'
).join('');
}
}
